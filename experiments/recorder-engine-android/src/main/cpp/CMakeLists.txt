cmake_minimum_required(VERSION 3.22.1)
project(onetake_qnn LANGUAGES CXX)
add_library(onetake_qnn_probe SHARED qnn_runtime_probe.cpp)
target_compile_features(onetake_qnn_probe PRIVATE cxx_std_17)
target_link_libraries(onetake_qnn_probe PRIVATE dl)

add_library(onetake_qnn_graph SHARED qnn_graph_session.cpp)
target_compile_features(onetake_qnn_graph PRIVATE cxx_std_17)
target_link_libraries(onetake_qnn_graph PRIVATE dl)

if(QAIRT_SDK_ROOT AND ANDROID_ABI STREQUAL "arm64-v8a")
    if(NOT EXISTS "${QAIRT_SDK_ROOT}/include/QNN/QnnInterface.h")
        message(FATAL_ERROR "QAIRT_SDK_ROOT does not contain QNN headers")
    endif()
    if(NOT EXISTS "${QAIRT_SDK_ROOT}/include/QNN/System/QnnSystemInterface.h")
        message(FATAL_ERROR "QAIRT_SDK_ROOT does not contain QNN System headers")
    endif()
    target_include_directories(onetake_qnn_probe PRIVATE "${QAIRT_SDK_ROOT}/include/QNN")
    target_compile_definitions(onetake_qnn_probe PRIVATE ONETAKE_QNN_AVAILABLE=1)
    target_include_directories(onetake_qnn_graph PRIVATE "${QAIRT_SDK_ROOT}/include/QNN")
    target_compile_definitions(onetake_qnn_graph PRIVATE ONETAKE_QNN_AVAILABLE=1)
endif()
