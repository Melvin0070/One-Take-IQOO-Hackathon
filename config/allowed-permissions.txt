# R17 — the single-phone demo build asks only for permissions it uses.
#
# `./gradlew guardPermissions` fails the build if the MERGED manifest contains a
# permission that is not on this list. Every line needs a reason. If you cannot
# write the reason, you cannot add the line.
#
# The infosec juror can read this file and the merged manifest side by side.
# That is the point.

android.permission.CAMERA                                   # the take
android.permission.RECORD_AUDIO                             # recognition + the cut
android.permission.FOREGROUND_SERVICE                       # capture survives navigation
android.permission.FOREGROUND_SERVICE_CAMERA                # required alongside camera FGS type
android.permission.FOREGROUND_SERVICE_MICROPHONE            # required alongside microphone FGS type
android.permission.FOREGROUND_SERVICE_MEDIA_PROCESSING      # export, foreground-only with visible progress

# Generated by androidx.core for its own non-exported dynamic receivers. Signature-level,
# scoped to this package, grants nothing to anyone. Not a capability the app asks for.
com.onetake.app.DYNAMIC_RECEIVER_NOT_EXPORTED_PERMISSION

# Removed in the app manifest with tools:node="remove", because Media3 declares them and
# we do not need them. Listed here so the next person knows it was deliberate:
#   android.permission.ACCESS_NETWORK_STATE  <- ExoPlayer bandwidth meter. P1 forbids it.
#   android.permission.WAKE_LOCK             <- ExoPlayer setWakeMode. We use FLAG_KEEP_SCREEN_ON.
